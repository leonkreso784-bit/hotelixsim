```ts
import { t } from "@hotelixsim/i18n";
import type { CauseCode, MonthCode, StrategyCode, UnitCode, VarCode } from "./api/types";

const nf2 = new Intl.NumberFormat("hr-HR", { maximumFractionDigits: 2 });
const nf0 = new Intl.NumberFormat("hr-HR", { maximumFractionDigits: 0 });
const MINUS = "−";

/** Indeks u stotinkama (0–10000) → tekst za prikaz, npr. 5025 → "50,25". */
export function formatIndex(hundredths: number): string {
  return nf2.format(hundredths / 100);
}

function signed(value: number, abs: string): string {
  if (value === 0) return abs;
  return value > 0 ? `+${abs}` : `${MINUS}${abs}`;
}

/** Stotinke s predznakom: 825 → "+8,25", -200 → "−2", 0 → "0". */
export function formatSigned(hundredths: number): string {
  return signed(hundredths, nf2.format(Math.abs(hundredths) / 100));
}

/** Centi → cijeli euri: 9 800 000 → "98.000 €". */
export function formatEuros(cents: number): string {
  const euros = Math.round(cents / 100);
  return `${euros < 0 ? MINUS : ""}${nf0.format(Math.abs(euros))} €`;
}

export function formatSignedEuros(cents: number): string {
  return signed(cents, formatEuros(Math.abs(cents)));
}

/** Ocjena u stotinkama na ljestvici do 10: 840 → "8,4/10". */
export function formatRating(hundredths: number): string {
  return `${nf2.format(hundredths / 100)}/10`;
}

export function formatOccupancy(sold: number, available: number): string {
  if (available <= 0) return "—";
  return `${Math.round((sold * 100) / available)} %`;
}

/** Iznos promjene s jedinicom iz obračuna (feed `change`). */
export function formatChange(amount: number, unit: UnitCode): string {
  switch (unit) {
    case "cents":
      return formatSignedEuros(amount);
    case "room_nights":
      return `${signed(amount, nf0.format(Math.abs(amount)))} ${t("unit.room_nights")}`;
    case "people":
      return `${signed(amount, nf0.format(Math.abs(amount)))} ${t("unit.people")}`;
    case "points":
      return `${formatSigned(amount)} ${t("unit.points")}`;
    case "rating_hundredths":
      return `${formatSigned(amount)} ${t("unit.rating_hundredths")}`;
    case "bp":
      return `${formatSigned(amount)} %`;
  }
}

export function monthName(code: MonthCode): string {
  return t(`month.${code}` as const);
}

export function capitalize(s: string): string {
  return s.charAt(0).toLocaleUpperCase("hr-HR") + s.slice(1);
}

export function causeLabel(code: CauseCode): string {
  return t(`cause.${code}` as const);
}

export function varLabel(code: VarCode): string {
  return t(`var.${code}` as const);
}

export function strategyLabel(code: StrategyCode): string {
  return t(`strategy.${code}` as const);
}
```

```ts
// Čista logika igre za ekrane: indeksi, promjene, ljestvica. Bez Reacta i mreže.
import type { GameBody, IndexCode, IndexEntry, MonthCode, PublicHotelBody, StrategyCode } from "./api/types";

export const INDEX_ORDER: readonly IndexCode[] = ["G", "R", "Z", "O", "F"];

export function sortIndices(xs: IndexEntry[]): IndexEntry[] {
  return [...xs].sort((a, b) => INDEX_ORDER.indexOf(a.code) - INDEX_ORDER.indexOf(b.code));
}

export function indicesAt(h: PublicHotelBody, month: MonthCode): IndexEntry[] | null {
  return h.months.find((m) => m.month === month)?.indices ?? null;
}

/** Zadnji obračunati mjesec, a prije prvog obračuna početno stanje. */
export function latestIndices(h: PublicHotelBody): IndexEntry[] {
  return h.months.at(-1)?.indices ?? h.start;
}

/** Indeksi prije zadanog mjeseca: prethodni mjesec igre ili početak. */
export function indicesBefore(h: PublicHotelBody, months: MonthCode[], month: MonthCode): IndexEntry[] {
  const i = months.indexOf(month);
  const prev = i > 0 ? months[i - 1] : undefined;
  return (prev && indicesAt(h, prev)) || h.start;
}

export function indexDeltas(from: IndexEntry[], to: IndexEntry[]): Map<IndexCode, number> {
  const before = new Map(from.map((e) => [e.code, e.hundredths]));
  return new Map(to.map((e) => [e.code, e.hundredths - (before.get(e.code) ?? e.hundredths)]));
}

export function progressAt(h: PublicHotelBody, month: MonthCode): number | null {
  return h.months.find((m) => m.month === month)?.progress_hundredths ?? null;
}

export type LeaderRow = {
  hotelId: string;
  name: string;
  mine: boolean;
  strategy: StrategyCode | null;
  progress: number;
};

/** Ljestvica napretka za mjesec: samo obračunati hoteli, najbolji prvi, jednaki po mjestu. */
export function leaderboard(game: GameBody, month: MonthCode): LeaderRow[] {
  const rows: (LeaderRow & { slot: number })[] = [];
  for (const h of game.hotels) {
    const progress = progressAt(h, month);
    if (progress === null) continue;
    rows.push({
      hotelId: h.hotel_id,
      name: h.name,
      mine: h.hotel_id === game.my_hotel_id,
      strategy: h.seat.strategy,
      progress,
      slot: h.slot,
    });
  }
  rows.sort((a, b) => b.progress - a.progress || a.slot - b.slot);
  return rows.map(({ slot: _slot, ...row }) => row);
}

export function monthPosition(game: GameBody): { month: MonthCode; index: number; total: number } | null {
  if (!game.current) return null;
  return {
    month: game.current.month,
    index: game.months.indexOf(game.current.month),
    total: game.months.length,
  };
}

export function nextMonth(months: MonthCode[], month: MonthCode): MonthCode | null {
  return months[months.indexOf(month) + 1] ?? null;
}
```

```tsx
export function Delta({ value, size = "sm" }: { value: number | null; size?: "sm" | "lg" }) {
```

```ts
// Izgled lika kao podatak (temelj za uređivač kao u BitLifeu, ADR-021/022). Lik se NIKAD ne crta mimo AvatarLook.
import { BACKDROP, CLOTHES, HAIR, SKIN } from "./palette";

export const HAIR_STYLES = ["none", "buzz", "short", "side", "curly", "long", "ponytail", "bun"] as const;
export const EYES = ["dot", "happy", "wide"] as const;
export const MOUTHS = ["smile", "neutral", "worried", "grin"] as const;
export const FACIAL_HAIR = ["none", "stubble", "mustache", "beard"] as const;
export const GLASSES = ["none", "round", "square"] as const;
export const HATS = ["none", "chef", "cap"] as const;

export type HairStyle = (typeof HAIR_STYLES)[number];
export type Eyes = (typeof EYES)[number];
export type Mouth = (typeof MOUTHS)[number];
export type FacialHair = (typeof FACIAL_HAIR)[number];
export type Glasses = (typeof GLASSES)[number];
export type Hat = (typeof HATS)[number];

export type AvatarLook = {
  skin: number;
  hair: HairStyle;
  hairColor: number;
  eyes: Eyes;
  mouth: Mouth;
  facialHair: FacialHair;
  glasses: Glasses;
  hat: Hat;
  clothes: number;
  backdrop: number;
};

const inRange = (i: number, n: number) => Number.isInteger(i) && i >= 0 && i < n;
const oneOf = <T extends string>(xs: readonly T[], v: string) => (xs as readonly string[]).includes(v);

export function isValidLook(l: AvatarLook): boolean {
  return (
    inRange(l.skin, SKIN.length) &&
    inRange(l.hairColor, HAIR.length) &&
    inRange(l.clothes, CLOTHES.length) &&
    inRange(l.backdrop, BACKDROP.length) &&
    oneOf(HAIR_STYLES, l.hair) &&
    oneOf(EYES, l.eyes) &&
    oneOf(MOUTHS, l.mouth) &&
    oneOf(FACIAL_HAIR, l.facialHair) &&
    oneOf(GLASSES, l.glasses) &&
    oneOf(HATS, l.hat)
  );
}

/** FNV-1a (32 bita) pa xorshift: isto ime → isti niz brojeva, na svakom računalu. */
function rng(seed: string): (n: number) => number {
  let h = 0x811c9dc5;
  for (let i = 0; i < seed.length; i++) h = Math.imul(h ^ seed.charCodeAt(i), 0x01000193) >>> 0;
  return (n: number) => {
    h ^= h << 13;
    h ^= h >>> 17;
    h ^= h << 5;
    h >>>= 0;
    return h % n;
  };
}

function pick<T>(xs: readonly T[], r: (n: number) => number): T {
  return xs[r(xs.length)] as T;
}

export function lookFromSeed(seed: string): AvatarLook {
  const r = rng(seed);
  const hair = pick(HAIR_STYLES, r);
  // Brada rjeđe od ostalog, kapa vrlo rijetko — „slučajni“ gosti trebaju izgledati obično.
  const facialHair = r(3) === 0 ? pick(FACIAL_HAIR, r) : "none";
  return {
    skin: r(SKIN.length),
    hair,
    hairColor: r(HAIR.length - 1), // sijeda (zadnja) samo ručno
    eyes: pick(EYES, r),
    mouth: pick(MOUTHS, r),
    facialHair,
    glasses: r(4) === 0 ? pick(GLASSES, r) : "none",
    hat: r(12) === 0 ? "cap" : "none",
    clothes: r(CLOTHES.length - 1), // bijela (kuhar) samo ručno
    backdrop: r(BACKDROP.length),
  };
}
```

```tsx
// Mali usmjerivač nad History API-jem (ADR-022 t. 1): sedam ruta, bez učitavača.
import { type AnchorHTMLAttributes, type MouseEvent, useSyncExternalStore } from "react";
import type { MonthCode } from "./api/types";

export type Route =
  | { name: "hotel" }
  | { name: "situations" }
  | { name: "situation"; code: string }
  | { name: "plan" }
  | { name: "month"; month: MonthCode }
  | { name: "feed" }
  | { name: "avatars" }
  | { name: "notFound" };

const MONTHS: readonly MonthCode[] = ["jun", "jul", "aug", "sep"];

function isMonth(s: string): s is MonthCode {
  return (MONTHS as readonly string[]).includes(s);
}

export function matchRoute(pathname: string): Route {
  const [a, b, ...rest] = pathname.split("/").filter(Boolean);
  if (rest.length > 0) return { name: "notFound" };
  if (a === undefined || (a === "demo" && b === undefined)) return { name: "hotel" };
  if (a === "situacije") return b === undefined ? { name: "situations" } : { name: "situation", code: decodeURIComponent(b) };
  if (a === "plan" && b === undefined) return { name: "plan" };
  if (a === "mjesec" && b !== undefined && isMonth(b)) return { name: "month", month: b };
  if (a === "tijek" && b === undefined) return { name: "feed" };
  if (a === "likovi" && b === undefined) return { name: "avatars" };
  return { name: "notFound" };
}

const listeners = new Set<() => void>();

function subscribe(onChange: () => void) {
  listeners.add(onChange);
  window.addEventListener("popstate", onChange);
  return () => {
    listeners.delete(onChange);
    window.removeEventListener("popstate", onChange);
  };
}

export function navigate(to: string): void {
  if (to !== window.location.pathname) window.history.pushState(null, "", to);
  for (const l of listeners) l();
  document.documentElement.scrollTop = 0;
}

export function useRoute(): Route {
  const path = useSyncExternalStore(subscribe, () => window.location.pathname);
  return matchRoute(path);
}

type LinkProps = AnchorHTMLAttributes<HTMLAnchorElement> & { to: string };

export function Link({ to, onClick, ...rest }: LinkProps) {
  function handle(e: MouseEvent<HTMLAnchorElement>) {
    onClick?.(e);
    if (e.defaultPrevented || e.button !== 0 || e.metaKey || e.ctrlKey || e.shiftKey || e.altKey) return;
    e.preventDefault();
    navigate(to);
  }
  return <a href={to} onClick={handle} {...rest} />;
}
```

```ts
import { isMessageKey, t } from "@hotelixsim/i18n";

/** Greška API-ja: HTTP status i stabilan kod problema (`Problem.code`) ako ga tijelo ima. */
export class ApiError extends Error {
  constructor(
    readonly status: number,
    readonly code: string | null,
  ) {
    super(`API ${status}${code ? ` ${code}` : ""}`);
    this.name = "ApiError";
  }
}

function problemCode(body: unknown): string | null {
  if (typeof body === "object" && body !== null && "code" in body && typeof body.code === "string") return body.code;
  return null;
}

/** Rezultat openapi-fetch poziva → podaci ili ApiError. */
export function unwrap<T>(r: { data?: T; error?: unknown; response: Response }): T {
  if (r.data !== undefined) return r.data;
  throw new ApiError(r.response.status, problemCode(r.error));
}

/** 4xx je odluka poslužitelja i ne ponavlja se; 5xx i mreža jednom. */
export function shouldRetry(failureCount: number, error: unknown): boolean {
  if (error instanceof ApiError && error.status < 500) return false;
  return failureCount < 1;
}

export function problemMessage(error: unknown): string {
  if (error instanceof ApiError) {
    const key = `problem.${error.code ?? ""}`;
    if (error.code && isMessageKey(key)) return t(key);
    return error.status >= 500 ? t("error.server") : t("error.generic");
  }
  return t("error.server");
}

export function isNoGame(error: unknown): boolean {
  return error instanceof ApiError && (error.status === 401 || error.status === 404);
}
```

```ts
// test/fakeApi.ts — lažni poslužitelj za testove: rute po metodi i putanji, zapis svih poziva.
export function json(status: number, body: unknown): Response {
  return new Response(JSON.stringify(body), { status, headers: { "content-type": "application/json" } });
}

export function problem(status: number, code: string): Response {
  return new Response(JSON.stringify({ type: "about:blank", title: code, status, code }), {
    status,
    headers: { "content-type": "application/problem+json" },
  });
}

type Params = Record<string, string>;
export type Handler = (req: Request, params: Params, body: unknown) => Response | Promise<Response>;

type RouteEntry = { method: string; pattern: RegExp; names: string[]; handler: Handler };

export class FakeApi {
  readonly calls: { method: string; path: string; body: unknown }[] = [];
  private routes: RouteEntry[] = [];

  /** `path` kao "/v1/hotels/:hotel/situations"; tijelo umjesto funkcije = 200 JSON. Kasnija ruta pobjeđuje. */
  on(method: string, path: string, handlerOrBody: Handler | object): this {
    const names: string[] = [];
    const pattern = new RegExp(
      `^${path.replace(/:(\w+)/g, (_m, name: string) => {
        names.push(name);
        return "([^/]+)";
      })}$`,
    );
    const handler: Handler =
      typeof handlerOrBody === "function" ? (handlerOrBody as Handler) : () => json(200, handlerOrBody);
    this.routes.unshift({ method, pattern, names, handler });
    return this;
  }

  posts(path: string) {
    return this.calls.filter((c) => c.method !== "GET" && c.path === path);
  }

  fetch = async (req: Request): Promise<Response> => {
    const url = new URL(req.url);
    const text = req.method === "GET" ? "" : await req.clone().text();
    const body: unknown = text ? JSON.parse(text) : undefined;
    this.calls.push({ method: req.method, path: url.pathname, body });
    for (const r of this.routes) {
      if (r.method !== req.method) continue;
      const m = r.pattern.exec(url.pathname);
      if (!m) continue;
      const params: Params = {};
      r.names.forEach((n, i) => {
        params[n] = decodeURIComponent(m[i + 1] ?? "");
      });
      return r.handler(req, params, body);
    }
    return problem(404, "unknown_route");
  };
}
```

```ts
export function useConversation(hotelId: string, code: string, enabled: boolean) {
  const api = useApi();
  return useQuery<ConversationBody>({
    queryKey: keys.conversation(hotelId, code),
    enabled,
    queryFn: async () =>
      unwrap(
        await api.GET("/v1/hotels/{hotel_id}/situations/{code}/conversation", {
          params: { path: { hotel_id: hotelId, code } },
        }),
      ),
    refetchInterval: (q) => (q.state.data?.awaiting_reply ? poll.conversationMs : false),
  });
}

export function useSendMessage(hotelId: string, code: string) {
  const api = useApi();
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (text: string) =>
      unwrap(
        await api.POST("/v1/hotels/{hotel_id}/situations/{code}/messages", {
          params: { path: { hotel_id: hotelId, code } },
          body: { text },
        }),
      ),
    onSuccess: (conv) => qc.setQueryData(keys.conversation(hotelId, code), conv),
  });
}

export function useAnswer(hotelId: string, code: string) {
  const api = useApi();
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (answer: number) =>
      unwrap(
        await api.POST("/v1/hotels/{hotel_id}/situations/{code}/answer", {
          params: { path: { hotel_id: hotelId, code } },
          body: { answer },
        }),
      ),
    onSuccess: (body) => {
      qc.setQueryData(keys.situations(hotelId), body);
      void qc.invalidateQueries({ queryKey: keys.feed(hotelId) });
    },
  });
}
```

```tsx
export function situationStatus(s: SituationBody): { text: string; tone: "accent" | "muted" | "up" } {
```

```tsx
function int(s: string, min: number, max: number): number | null {
  if (!/^\d+$/.test(s.trim())) return null;
  const n = Number(s.trim());
  return n >= min && n <= max ? n : null;

function toPlan(d: Draft, l: PlanLimitsBody): PlanBody | null {
```

```ts
export function useGameUntilResolved(gameId: string, hotelId: string, month: MonthCode) {
  const api = useApi();
  return useQuery<GameBody>({
    queryKey: keys.game(gameId),
    queryFn: async () => unwrap(await api.GET("/v1/games/{game_id}", { params: { path: { game_id: gameId } } })),
    // Ispituje samo dok je baš taj mjesec trenutni a moj hotel ga još nema obračunatog (zatvaranje ga prebaci u closing).
    refetchInterval: (q) => {
      const g = q.state.data;
      const mine = g?.hotels.find((h) => h.hotel_id === hotelId);
      if (mine?.months.some((m) => m.month === month)) return false;
      return g?.current?.month === month ? poll.roundMs : false;
    },
  });
}

export function useFeed(hotelId: string, opts: { untilMentorFor?: MonthCode } = {}) {
  const api = useApi();
  return useQuery<FeedItemBody[]>({
    queryKey: keys.feed(hotelId),
    queryFn: async () =>
      unwrap(await api.GET("/v1/hotels/{hotel_id}/feed", { params: { path: { hotel_id: hotelId } } })).items,
    refetchInterval: (q) => {
      const m = opts.untilMentorFor;
      if (!m) return false;
      const has = q.state.data?.some((i) => i.kind === "mentor" && i.month === m);
      return has || q.state.dataUpdateCount >= poll.maxFeedPolls ? false : poll.feedMs;
    },
  });
}
```

```tsx
import { t } from "@hotelixsim/i18n";
import { problemMessage } from "../api/errors";
import { useFeed } from "../api/queries";
import type { FeedItemBody, HotelBody, MonthCode } from "../api/types";
import { capitalize, causeLabel, formatChange, formatSigned, monthName, varLabel } from "../format";
import { Card, ErrorText, StatusText } from "../ui/primitives";
import type { FeedChange } from "./month/Changes";

type Plain = Exclude<FeedItemBody, FeedChange>;
export type FeedEntry =
  | { key: string; month: MonthCode | null; kind: "item"; item: Plain }
  | { key: string; month: MonthCode; kind: "changes"; changes: FeedChange[] };

export function feedEntries(items: FeedItemBody[]): FeedEntry[] {
  const out: FeedEntry[] = [];
  for (const i of items) {
    if (i.kind === "change") {
      const last = out.at(-1);
      if (last?.kind === "changes" && last.month === i.month) last.changes.push(i);
      else out.push({ key: `c${i.seq}`, month: i.month, kind: "changes", changes: [i] });
    } else {
      out.push({ key: `i${i.seq}`, month: "month" in i ? i.month : null, kind: "item", item: i });
    }
  }
  return out.reverse();
}

function sentence(i: Plain): string {
  switch (i.kind) {
    case "hotel_received":
      return t("feed.hotel_received", { name: i.name });
    case "plan_changed":
      return t(i.by === "robot" ? "feed.plan_changed.robot" : "feed.plan_changed", { month: monthName(i.month) });
    case "answer_chosen":
      return t("feed.answer_chosen", { title: i.title, label: i.label });
    case "round_closed":
      return t("feed.round_closed", { month: capitalize(monthName(i.month)) });
    case "situation_ignored":
      return t("feed.situation_ignored", { title: i.title });
    case "month_resolved":
      return t("feed.month_resolved", { month: capitalize(monthName(i.month)), progress: formatSigned(i.progress_hundredths) });
    case "response":
      return t("feed.response", { title: i.title });
    case "verdict":
      return t(i.late ? "feed.verdict.late" : i.neutral ? "feed.verdict.neutral" : "feed.verdict", {
        title: i.title,
        q: i.quality,
      });
    case "mentor":
      return t("feed.mentor", { text: i.text });
  }
}

export function FeedScreen({ me }: { me: HotelBody }) {
  const feed = useFeed(me.hotel.hotel_id);
  if (feed.isPending) return <StatusText>{t("status.loading")}</StatusText>;
  if (feed.isError) return <ErrorText>{problemMessage(feed.error)}</ErrorText>;
  const entries = feedEntries(feed.data);
  const groups: { month: MonthCode | null; entries: FeedEntry[] }[] = [];
  for (const e of entries) {
    const g = groups.at(-1);
    if (g && g.month === e.month) g.entries.push(e);
    else groups.push({ month: e.month, entries: [e] });
  }
  return (
    <>
      <h1 className="text-3xl font-extrabold tracking-tight">{t("feed.title")}</h1>
      {entries.length === 0 && <p className="text-muted">{t("feed.empty")}</p>}
      {groups.map((g) => (
        <section key={`${g.month ?? "start"}-${g.entries[0]?.key}`} className="space-y-2">
          <h2 className="pt-2 text-sm font-bold tracking-wide text-muted uppercase">
            {g.month ? capitalize(monthName(g.month)) : t("feed.start")}
          </h2>
          <Card className="divide-y divide-line py-1">
            <ol>
              {g.entries.map((e) =>
                e.kind === "item" ? (
                  <li key={e.key} className="py-2">
                    {sentence(e.item)}
                  </li>
                ) : (
                  <li key={e.key} className="py-2">
                    <details>
                      <summary className="min-h-11 cursor-pointer font-semibold">
                        {t("feed.changes", { n: e.changes.length })}
                      </summary>
                      <ul className="text-sm">
                        {e.changes.map((c) => (
                          <li key={c.seq} className="flex justify-between gap-3 py-0.5">
                            <span className="text-muted">
                              {varLabel(c.var)} · {causeLabel(c.cause)}
                            </span>
                            <span className="tabular-nums">{formatChange(c.amount, c.unit)}</span>
                          </li>
                        ))}
                      </ul>
                    </details>
                  </li>
                ),
              )}
            </ol>
          </Card>
        </section>
      ))}
    </>
  );
}
```
